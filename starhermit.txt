# StarHermit engine script game: one package, played natively by the
# StarHermit client and in browsers by the engine's web runtime. Browsers open
# launch=index.html, the loader `npm run package` writes (it pins the engine
# on wiki.starhermit.com by hash; never edit it). Re-run `npm run package`
# after every change and commit index.html + starhermit-files.json.
launch=index.html
engine.api=1
engine.entry=native/main.ts
engine.version=1.18.0
engine.minor=19
engine.capabilities=platform,storage,clipboard

# Keyboard controls (players rebind keys and pads in the runtime's Options).
# Touch screens drive with the on-screen pedal tray (GAS, BRAKE, TILT ◀ ▶).
control.throttle=ArrowUp+KeyW | Gas
control.brake=ArrowDown+KeyS | Brake
control.tilt_back=ArrowLeft+KeyA | Tilt back
control.tilt_forward=ArrowRight+KeyD | Tilt forward
control.pause=Escape+KeyP | Pause / resume
control.undo=KeyU | Undo (Practice)
control.recenter=KeyC | Re-centre camera
engine.group.throttle=Driving
engine.group.pause=Run

# Gamepad defaults (the browser build's standard mapping: RT gas, LT brake,
# the left stick tilts, Start pauses), plus the D-pad for tilting, X to undo
# and the right stick press to re-centre the camera.
engine.gamepad.throttle=RightTrigger
engine.gamepad.brake=LeftTrigger
engine.gamepad.tilt_back=LeftStickLeft+DpadLeft
engine.gamepad.tilt_forward=LeftStickRight+DpadRight
engine.gamepad.pause=Start
engine.gamepad.undo=West
engine.gamepad.recenter=RightStickPress
